--@name Safe Trace
--@author legokidlogan
--@shared

--[[
    - Because Starfall now errors when having an invalid entity in a trace filter,
       tedious jank like this is required.
    - Thanks for the code bloat and perf hit, Sparky.
    - Note that the original trace.line and trace.hull functions do NOT get localized or wrapped here.
    - WARNING: All of these functions will modify the original filter table if it has invalid entities!
--]]


local isValid = isValid
local tableRemove = table.remove
local isfunction = isfunction
local pcall = pcall


-- Only operates on sequential tables.
function table.removeInvalid( tbl )
    for i = #tbl, 1, -1 do
        if not isValid( tbl[i] ) then
            tableRemove( tbl, i )
        end
    end

    return tbl
end
local tableRemoveInvalid = table.removeInvalid

function trace.validateFilter( filter )
    if not filter then return nil end
    if isfunction( filter ) then return filter end
    if filter.entOwner then return isValid( filter ) and filter or nil end -- Validate single entity. No one should be passing non-sequential tables, so this check is sufficient.

    return tableRemoveInvalid( filter )
end
local traceValidateFilter = trace.validateFilter

-- Auto-validates (potentially modifying) filter. O(n^2) due to item removal.
function trace.lineSafe( start, endpos, filter, ... )
    return trace.line( start, endpos, traceValidateFilter( filter ), ... )
end

-- Auto-validates (potentially modifying) filter. O(n^2) due to item removal.
function trace.hullSafe( start, endpos, minbox, maxbox, filter, ... )
    return trace.full( start, endpos, minbox, maxbox, traceValidateFilter( filter ), ... )
end

--[[
    - Same as trace.lineSafe, but tries an unvalidated pcall first, which
       only adds O(1) perf cost to trace.line if everything in filter is valid.
    - Ideal for setups where you expect invalid entities to be possible but rare, and you re-use the same table.
    - Do not use this if invalid entities are common, or if a new filter table is made for each call.
--]]
function trace.lineSafeStable( start, endpos, filter, ... )
    local success, tr = pcall( trace.line, start, endpos, filter, ... )
    if success then return tr end

    return trace.line( start, endpos, traceValidateFilter( filter ), ... )
end

-- Same as trace.lineSafeStable, but for hull traces.
function trace.hullSafeStable( start, endpos, minbox, maxbox, filter, ... )
    local success, tr = pcall( trace.hull, start, endpos, minbox, maxbox, filter, ... )
    if success then return tr end

    return trace.hull( start, endpos, minbox, maxbox, traceValidateFilter( filter ), ... )
end
